ភាពជូរចត់ដែលផ្អែមល្ហែម
១៥ ធ្នូ ២០២៤

ចំពោះអ្នកជឿស្មោះត្រង់ក្នុងព្រះគ្រីស្ទ៖ កុំភ្ញាក់ផ្អើលឡើយ!

កុំភ្ញាក់ផ្អើលឡើយ បងប្អូនអើយ កុំភ្ញាក់ផ្អើលឡើយ ទោះបីជាលោកីយ៍នេះស្អប់អ្នកក៏ដោយ។ ចូរដឹងថា លោកីយ៍នេះបានស្អប់ខ្ញុំមុនពេលដែលវាស្អប់អ្នកទៅទៀត ពួកស្ងួនភ្ងារបស់ខ្ញុំ ហើយអ្នកបម្រើមិនធំជាងចៅហ្វាយរបស់ខ្លួនឡើយ។ ដូច្នេះ ចូរយល់ពីពេលវេលាដែលអ្នកកំពុងរស់នៅ ហើយចូលឲ្យកាន់តែខិតជិតនៅក្នុងចិត្តរបស់អ្នកទៅរកព្រះយេស៊ូវ ហាម៉ាស៊ីអា (Yeshua HaMashiach) ពីព្រោះខ្ញុំកំពុងហៅអ្នកឲ្យខិតចូលមកជិតខ្ញុំរឹតតែខ្លាំង ឱពួកជំនុំរបស់ខ្ញុំ រូបកាយខ្ញុំ និងសូម្បីតែកូនក្រមុំរបស់ខ្ញុំ ហើយខ្ញុំកំពុងហៅអ្នកឲ្យកាន់តែជិតស្និទ្ធក្នុងគ្រានេះ មុនពេលដែលអ្នកត្រូវបានលើកឡើងទៅស្ថានសួគ៌ ដើម្បីឲ្យអ្នកអាចឈ្នះគ្រប់ការទាំងអស់នៅក្នុងព្រះគ្រីស្ទ។

ហើយយ៉ាងទាន់ហន់និងដោយនឹកស្មានមិនដល់ ឥឡូវនេះអ្នកនឹងត្រូវបានយកទៅស្ថានសួគ៌ ពីព្រោះខ្ញុំមកនៅម៉ោងដែលអ្នកមិននឹកស្មានដល់។ មើលចុះ! ខ្ញុំមកដូចជាចោរនៅពេលយប់! ប៉ុន្តែ ដូចជានៅក្នុងជំនាន់លោកណូអេ នោះការយាងមករបស់កូនមនុស្សក៏នឹងដូច្នោះដែរ។ ដ្បិតដូចជានៅគ្រាមុនទឹកជំនន់ គេបានស៊ីផឹក រៀបការ និងលើកឲ្យមានប្ដីប្រពន្ធ រហូតដល់ថ្ងៃដែលលោកណូអេបានចូលទៅក្នុងទូកធំ ហើយគេមិនបានដឹងអ្វីសោះឡើយ ទាល់តែទឹកជំនន់បានមកដល់ ហើយយកគេទាំងអស់ទៅ នោះការយាងមករបស់កូនមនុស្សក៏នឹងដូច្នោះដែរ។

ដូច្នេះ ខ្ញុំបានទូន្មានអ្នក ពួកស្ងួនភ្ងារបស់ខ្ញុំ ឲ្យចាកចេញពីកង្វល់នៃលោកីយ៍នេះ ដោយដឹងថាការចាកចេញរបស់អ្នកពីលោកីយ៍ដ៏អាក្រក់នេះគឺជិតមកដល់ហើយ។ ប៉ុន្តែ ទោះជាយ៉ាងណា អ្នកនៅតែមានជីវិតដែលត្រូវរស់នៅ ហើយអ្នកត្រូវតែបន្តរហូតដល់អ្នកលែងមានជីវិត ពីព្រោះនៅពេលដែលអ្នកត្រូវបានយកទៅស្ថានសួគ៌ដោយនឹកស្មានមិនដល់តាមរយៈព្រះហស្តដ៏មានចេស្តានៃសេចក្តីសង្គ្រោះរបស់ខ្ញុំ ដំណើររបស់អ្នកនឹងត្រូវបញ្ចប់ ហើយជីវិតនៅលើផែនដីរបស់អ្នកក៏នឹងត្រូវបញ្ចប់នៅពេលនោះ។ ដូច្នេះ ខ្ញុំទូន្មានអ្នក ឱកូនតូចៗរបស់ខ្ញុំអើយ ចូររស់នៅដោយស្តាប់បង្គាប់ទាំងស្រុងចំពោះព្រះសូរសៀងរបស់ព្រះអម្ចាស់ និងជាព្រះអង្គសង្គ្រោះរបស់អ្នក គឺព្រះយេស៊ូវគ្រីស្ទជាព្រះអម្ចាស់ដែលបានយាងឡើងរបស់អ្នក ហើយខ្ញុំនឹងធ្វើឲ្យអ្នកទទួលបានជ័យជម្នះទាំងស្រុង!

ហេតុដូច្នេះ ពេលដែលអ្នករស់នៅ ចូរថ្វាយអ្វីៗទាំងអស់មកខ្ញុំនៅក្នុងការប្តេជ្ញាចិត្តនិងការលើកទឹកចិត្តក្នុងដួងចិត្តរបស់អ្នក ព្រោះខ្ញុំគឺជាភាពគ្រប់គ្រាន់របស់អ្នកក្នុងគ្រប់កិច្ចការទាំងអស់។ កុំគិតថាចង់និយាយនៅក្នុងចិត្តរបស់អ្នកថា អ្នកជោគជ័យ និងមានភាពចម្រុងចម្រើនដោយសារកម្លាំងដៃរបស់អ្នកផ្ទាល់ ឬដោយសារសញ្ញាបត្រពីស្ថាប័នដែលអ្នកជាច្រើនឲ្យតម្លៃយ៉ាងល្ងង់ខ្លៅលើសពីចំណេះដឹងរបស់ព្រះយេហូវ៉ា (Yahweh) នោះឡើយ សូម្បីតែសេចក្តីកុហកដែលបានលក់ឲ្យអ្នកជាសេចក្តីពិតក្នុងការអប់រំក្លែងក្លាយរបស់អ្នក ដែលត្រូវបានស្ថាបនាឡើងទាំងស្រុងលើសេចក្តីកុហក ប៉ុន្តែអ្នកបែរជាអួតយ៉ាងល្ងង់ខ្លៅពីសមត្ថភាពដែលអ្នកទទួលបានដោយសារការទាំងនោះ។ ដូច្នេះ ខ្ញុំប្រាប់អ្នកម្តងទៀត ឲ្យចេញពីចំណោមពួកគេ ហើយញែកខ្លួនចេញក្នុងគ្រប់កិច្ចការទាំងអស់ ឬអ្នកនឹងទទួលចំណែកក្នុងការជំនុំជម្រះនិងសេចក្តីវិនាសរបស់ពួកគេ នៅពេលដែលរឿងបែបនេះមិនចាំបាច់សោះ ព្រោះនៅគ្រប់ពេលវេលាទាំងអស់ឥឡូវនេះ ខ្ញុំកំពុងហៅ និងអង្វរអ្នកឲ្យមករកខ្ញុំទាំងស្រុងដោយសេចក្តីស្រឡាញ់របស់ខ្ញុំចំពោះអ្នក កូនៗរបស់ខ្ញុំ។ ដូច្នេះ ចូរចាកចេញពីសេចក្តីអំនួតដ៏គ្មានតម្លៃដែលចំណេះដឹងនាំមកចុះ!

ហេតុនេះហើយ ខ្ញុំឲ្យអ្នកយល់ថា អ្វីៗដែលអ្នកទទួលបាននៅក្នុងជីវិតនេះទាក់ទងនឹងទ្រព្យសម្បត្តិលោកីយ៍ គឺត្រូវបានផ្តល់ឲ្យដោយសេចក្តីសព្វព្រះទ័យ សេចក្តីសប្បុរស និងព្រះគុណរបស់ព្រះយេហូវ៉ា ព្រះដ៏មាននាមថា "ខ្ញុំជា" ដែលទ្រង់សព្វព្រះទ័យលើសពីអ្វីៗទាំងអស់ឲ្យអ្នកបានចម្រុងចម្រើន និងមានសុខភាពល្អ ដូចជាព្រលឹងរបស់អ្នកបានចម្រុងចម្រើនដែរ។ ប៉ុន្តែ កុំល្ងង់ខ្លៅគិតថាអ្នកបង្កើតខ្លួនឯង ឬថានៅក្នុងការសរសើរតម្កើងខាងលោកីយ៍របស់អ្នក អ្នកត្រូវបានទទួលស្គាល់នៅចំពោះព្រះភ័ក្ត្រព្រះយេហូវ៉ា ព្រះដ៏មាននាមថា "ខ្ញុំជា" ឬព្រះយេស៊ូវគ្រីស្ទឡើយ ពីព្រោះគ្មានសាច់ឈាមណាត្រូវបានរាប់ជាសុចរិតនៅចំពោះភ្នែកយើងឡើយ។ ដូច្នេះ ចូរយល់ថា អ្នកដែលលើកតម្កើងខ្លួនឯងតាមរបៀបនេះ នឹងរងការខាតបង់យ៉ាងធំធេងដោយសារតែសេចក្តីអំនួតរបស់ខ្លួន ដែលប្រាកដជានាំទៅរកសេចក្តីវិនាសដ៏ធំ។
