អ្នកគង្វាលដ៏ស្មោះត្រង់មានបន្ទូលថា

អ្នកគង្វាលដ៏ស្មោះត្រង់មានបន្ទូលទៅកាន់អ្នកស្មោះត្រង់ថា៖ ជាប្រាកដមែន ខ្ញុំប្រាប់អ្នករាល់គ្នាថា ចូរថយចេញពីអំពើទុច្ចរិត ហើយកុំធ្វើបាបទៀតឡើយ។ ដ្បិតរង្វាន់របស់ខ្ញុំនៅជាមួយខ្ញុំ ដើម្បីតបស្នងដល់មនុស្សម្នាក់ៗតាមការប្រព្រឹត្តរបស់គេ។ ខ្ញុំបានហៅអ្នករាល់គ្នាក្នុងសេចក្តីស្រឡាញ់នៃព្រះវរបិតារបស់ខ្ញុំ ដើម្បីឲ្យអ្នករាល់គ្នាមានជីវិត ហើយមានជីវិតអស់កល្បជានិច្ច។ ខ្ញុំបានហៅអ្នករាល់គ្នាឲ្យដើរក្នុងនគររបស់ខ្ញុំ បម្រើខ្ញុំ និងរស់នៅលើផែនដីដោយសេចក្តីសុចរិត។ ខ្ញុំបានប្រទានព្រះគុណរបស់ខ្ញុំដល់អ្នករាល់គ្នា ដើម្បីឲ្យអ្នករាល់គ្នាចម្រើនឡើងនៅក្នុងខ្ញុំ ជាព្រះអម្ចាស់យេស៊ូវគ្រីស្ទរបស់អ្នករាល់គ្នា។ កុំរសាយចិត្តនឹងការប្រដៅរបស់ខ្ញុំឡើយ ដ្បិតអស់អ្នកណាដែលខ្ញុំស្រឡាញ់ នោះខ្ញុំក៏បន្ទោសអ្នកនោះដែរ ដើម្បីឲ្យព្រះគុណសង្គ្រោះរបស់ខ្ញុំអាចចាក់ឫសយ៉ាងជ្រៅនៅក្នុងខ្លួនអ្នករាល់គ្នា ដើម្បីឲ្យអ្នករាល់គ្នាអាចធ្វើការដ៏អស្ចារ្យ ការនៃព្រះពរ ការប្រោសឲ្យជា និងឲ្យអ្នករាល់គ្នាបង្កើតផលជាច្រើននៅក្នុងខ្ញុំ ឱរាស្ត្ររបស់ខ្ញុំអើយ។

ដ្បិតព្រះវរបិតារបស់ខ្ញុំជាព្រះនៃសេចក្តីស្រឡាញ់ ហើយសេចក្តីស្រឡាញ់របស់ព្រះវរបិតាខ្ញុំមិនអាចត្រូវយកឈ្នះបានឡើយ។ នៅក្នុងខ្ញុំ ជាព្រះអម្ចាស់ដ៏សុចរិត មាននូវភាពពោរពេញទាំងអស់ដែលព្រះវរបិតាបានប្រទានមកខ្ញុំ។ ទ្រង់បានប្រគល់គ្រប់ទាំងអស់មកក្នុងដៃខ្ញុំ ហើយដូចដែលផារ៉ោនបានតម្កើងយ៉ូសែប ហើយប្រគល់ស្រុកអេស៊ីបទាំងមូលទៅក្នុងដៃគាត់យ៉ាងណា ព្រះវរបិតាខ្ញុំក៏បានតម្កើងខ្ញុំ ហើយប្រគល់គ្រប់សេចក្តីទាំងនៅស្ថានសួគ៌ និងនៅលើផែនដីមកក្នុងដៃខ្ញុំយ៉ាងនោះដែរ។ ទ្រង់បានប្រទានព្រះនាមមួយដល់ខ្ញុំដែលខ្ពស់លើសជាងនាមទាំងពួង ដើម្បីឲ្យគ្រប់ទាំងជង្គង់ទន់ចុះនៅចំពោះព្រះនាមព្រះយេស៊ូវ ទាំងនៅស្ថានសួគ៌ និងលើផែនដី ហើយគ្រប់ទាំងអណ្តាតនឹងទទួលស្គាល់ភាពជាព្រះអម្ចាស់របស់ខ្ញុំ ទាំងនៅស្ថានសួគ៌ និងលើផែនដី។

ខ្ញុំបានប្រទានជីវិតអស់កល្បជានិច្ចដល់អ្នករាល់គ្នា ហើយខ្ញុំបានរៀបចំផ្លូវមួយសម្រាប់អ្នករាល់គ្នា ដើម្បីរត់គេចពីទីរហោស្ថាន និងគេចពីថ្ងៃនៃសេចក្តីក្រោធដ៏ធំដែលកំពុងតែមកដល់ ដូច្នេះកុំរសាយចិត្តឡើយ! កុំឲ្យនឿយហត់ក្នុងការធ្វើអំពើល្អឡើយ តែចូរឈរនៅចំពោះមុខខ្ញុំ នោះខ្ញុំនឹងបន្តទាញអ្នករាល់គ្នាចូលមកក្នុងវត្តមានរបស់ខ្ញុំ។ អ្នករាល់គ្នានឹងអង្គុយនៅស្ថានសួគ៌ ហើយនឹងចូលរួមក្នុងបល្ល័ង្ករបស់ខ្ញុំ ដូចដែលខ្ញុំបានចូលរួមក្នុងបល្ល័ង្កព្រះវរបិតារបស់ខ្ញុំដែរ។ ខ្ញុំមានមកុដជាច្រើន និងព្រះពរជាច្រើនដែលបានរក្សាទុកសម្រាប់អ្នករាល់គ្នា ជាប្រជារាស្ត្រដ៏ស្មោះត្រង់របស់ខ្ញុំ។ ចូរឈរនៅចំពោះមុខខ្ញុំ ហើយបានល្អឥតខ្ចោះនៅក្នុងខ្ញុំ។ ចូរងាកចេញពីអំពើបាបរបស់អ្នក ហើយបណ្តោយឲ្យពេលវេលានៃភាពស្រស់ស្រាយបក់ផ្លុំអង្កាមចេញពីជីវិតអ្នករាល់គ្នា។ នៅក្នុងខ្ញុំមានជីវិតអស់កល្បជានិច្ច ហើយនៅក្នុងខ្ញុំមានការបង្កើតថ្មី ជាមនុស្សថ្មី គឺព្រះគ្រីស្ទនៅក្នុងអ្នករាល់គ្នា ជាសេចក្តីសង្ឃឹមនៃសិរីល្អ! ចូរងាកចេញពីសេចក្តីអាក្រក់ ហើយងាកចេញពីអំពើបាប។ ចូរបន្តប្រកាសពីខ្ញុំនៅក្នុងចំណោមពួកសាសន៍ដទៃ ដ្បិតនៅមានមនុស្សដែលត្រូវសង្គ្រោះនៅក្នុងម៉ោងចុងក្រោយនេះ។ ចូរបន្តប្រកាសព្រះនាមខ្ញុំ ដ្បិតគ្មាននាមណាផ្សេងទៀតនៅក្រោមមេឃ ដែលមនុស្សអាចទទួលបានសេចក្តីសង្គ្រោះឡើយ។ ចូរទទួលយកការកម្សាន្តចិត្តនៅក្នុងសេចក្តីសម្រាកដែលខ្ញុំបានប្រទានឲ្យអ្នក ហើយដាក់បន្ទុករបស់អ្នករាល់គ្នានៅក្បែរជើងខ្ញុំ ដ្បិតបន្ទុករបស់ខ្ញុំងាយ ហើយនឹមរបស់ខ្ញុំក៏ស្រាល។

សម្រាប់អ្នកស្មោះត្រង់ (អ្នកដែលបានឈ្នះអំពើបាប និងសេចក្តីលំបាកដោយសារព្រះលោហិតលោះបាបរបស់ខ្ញុំ) ខ្ញុំប្រាប់អ្នកស្មោះត្រង់ថា៖ ខ្ញុំមានរង្វាន់ដ៏ធំដែលបានរក្សាទុកសម្រាប់អ្នករាល់គ្នានៅក្នុងដំណាក់ព្រះវរបិតាខ្ញុំ។ ភ្នែកមិនដែលឃើញ ត្រចៀកមិនដែលឮ ហើយចិត្តមនុស្សក៏មិនដែលនឹកគិតដល់សេចក្តីទាំងឡាយណាដែលបានរៀបចំសម្រាប់ពួកគេនោះឡើយ។ ខ្ញុំនឹងផ្តល់រង្វាន់យ៉ាងបរិបូរណ៍ដល់អ្នករាល់គ្នាដោយមកុដជាច្រើន ដូច្នេះចូរឈរសម្រាប់ខ្ញុំ ហើយស្មោះត្រង់នឹងសំឡេងរបស់ខ្ញុំ និងព្រះបន្ទូលរបស់ខ្ញុំ ដ្បិតនៅក្នុងព្រះបន្ទូលទាំងនោះ អ្នករាល់គ្នានឹងរកឃើញជីវិត។ តើខ្ញុំមិនមែនជាព្រះបន្ទូលដែលត្រឡប់ជាសាច់ឈាម ហើយសណ្ឋិតនៅកណ្តាលចំណោមអ្នករាល់គ្នាទេឬ?

ខ្ញុំជាអ្នកគង្វាលល្អ ដែលគង្វាលហ្វូងចៀម។

ខ្ញុំជាសេចក្តីប្រោសឲ្យជារបស់អ្នករាល់គ្នា ហើយខ្ញុំនឹងប្រោសអ្នករាល់គ្នាឲ្យជាសះស្បើយយ៉ាងពេញលេញ។

ខ្ញុំកើតពីស្ត្រីម្នាក់ ហើយត្រឡប់ជាសាច់ឈាម ទោះបីជាគ្មានអំពើបាបនៅក្នុងខ្ញុំក៏ដោយ។

ខ្ញុំត្រូវគេឆ្កាងសម្រាប់អំពើបាបរបស់អ្នក ហើយជាការពិតណាស់ គ្មាននរណាម្នាក់ដកយកជីវិតខ្ញុំឡើយ តែខ្ញុំបានលះបង់ជីវិតដោយស្ម័គ្រចិត្ត ដោយសារសេចក្តីស្រឡាញ់ដែលខ្ញុំមានចំពោះបុត្រាបុត្រីរបស់ព្រះវរបិតាខ្ញុំ។
