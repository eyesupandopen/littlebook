សំឡេងនៃផ្គរលាន់ទាំងប្រាំពីរវាយប្រហារ!!!

ចូរស្ដាប់ ចូរស្ដាប់ព្រះបន្ទូលរបស់ យេស៊ូវ ហាម៉ាស៊ីអា (Yeshua HaMashiach) ដែលឥឡូវនេះត្រូវបានប្រទានដោយព្រះរាជបញ្ជារបស់ អដូណាយ-យ៉ូវ៉េ (Adonai-YHVH) ដើម្បីបញ្ចេញទៅកាន់ផែនដី ជាមួយនឹងការធ្វើឱ្យសកម្មពេញលេញ និងពោរពេញដោយសេចក្តីក្រោធ តាមរយៈបញ្ជាដ៏ខ្លាំងក្លា ដើម្បីឱ្យអ្នករាល់គ្នាបានយល់ថា ការជំនុំជម្រះនៃថ្ងៃដ៏អស្ចារ្យនោះបានមកដល់លើផែនដីហើយ ឥឡូវនេះ!!!

សំឡេងនៃផ្គរលាន់ទាំងប្រាំពីរវាយប្រហារ!!!

ចូរស្ដាប់ ចូរស្ដាប់ព្រះបន្ទូលរបស់ យេស៊ូវ ហាម៉ាស៊ីអា ដែលឈរយ៉ាងពេញលេញជា សិង្ហនៃកុលសម្ព័ន្ធយូដា ដែលអ្នករាល់គ្នានឹងដឹងនៅមួយម៉ោងជាមួយនឹងសត្វសាហាវនោះ អំពីអ្វីដែលនឹងកើតឡើងនៅក្នុងថ្ងៃដ៏អស្ចារ្យ និងច្របូកច្របល់នោះ មកលើសហរដ្ឋអាមេរិកទាំងនេះ នូវភាពធ្ងន់ធ្ងរដាច់ខាតនៃសេចក្តីក្រោធរបស់កូនចៀមនៃ អដូណាយ-យ៉ូវ៉េ ដែលនឹងបោកបក់យ៉ាងហិង្សាមកលើអ្នករាល់គ្នា និងជាពិសេសមកលើផែនដី។ នៅពេលដែលខ្ញុំ យេស៊ូវ ហាម៉ាស៊ីអា ដែលជាព្រះយេស៊ូវគ្រីស្ទផងដែរ ឈរឡើងដើម្បីបើកត្រានៅលើក្រាំងមរតករបស់ខ្ញុំ លើកលែងតែអ្វីដែលនឹងមកដល់នៅមួយម៉ោងជាមួយនឹងសត្វសាហាវនោះ វានឹងធ្វើឱ្យអ្នករាល់គ្នាដែលជាអ្នករស់នៅលើផែនដីភ័យខ្លាចយ៉ាងខ្លាំង នៅក្នុងអំណាចដ៏ពេញលេញ ហិង្សា ក្រោធខឹង និងការដាក់ទោស ដែលប្រទានពីដៃរបស់ខ្ញុំទាំងស្រុង ទៅក្នុងនគរដ៏គ្មានតម្លៃទាំងមូលរបស់ ហេលេល បេន សាហា (Heylel Ben Shachar) ដើម្បីឱ្យពួកគេនៅស្ងៀមស្ងាត់ ហើយយល់ពីអំណាច និងឫទ្ធានុភាពដ៏ក្រោធខឹងរបស់ខ្ញុំ នៅក្នុងភាពភ័យរន្ធត់ជាទីបំផុត!!!

ពីព្រោះពេលនោះ អ្នកនឹងឃើញ ហើយយល់នៅក្នុងភាពភ័យរន្ធត់ដ៏ធំធេង និងការភ័យខ្លាចដ៏អស់កល្បជានិច្ច អំពីអ្វីដែលនឹងតាមប្រមាញ់អ្នករាល់គ្នា ហើយបញ្ចាំងបំពានអ្នករាល់គ្នានៅក្នុងសេចក្តីក្រោធដ៏ពេញលេញ និងហិង្សា ដោយដៃរបស់ ប៊ីប្លារីឌាន១៤៤ (Biblaridion144) និងសូម្បីតែពួក ១៤៤,០០០ របស់ខ្ញុំទាំងអស់ ដែលនឹងអនុវត្តការជំនុំជម្រះដ៏សុចរិត និងពេញលេញចំពោះអ្នករាល់គ្នាដែលបានធ្លាក់ចុះ ចាប់ពីអ្នកតូចបំផុតរហូតដល់អ្នកធំបំផុត ជាពិសេសក្នុងអំឡុងពេលដែលអ្នកចាត់ទុកថាជាយុគសម័យមាសរបស់អ្នក!

ពីព្រោះអ្នករាល់គ្នាត្រូវបានប្រាប់ហើយថា នឹងគ្មានយុគសម័យមាសសម្រាប់អ្នកទេ ប៉ុន្តែជំនួសមកវិញគឺការជំនុំជម្រះដ៏ឃោរឃៅ។ ដូច្នេះ ដោយអំណាច និងឫទ្ធានុភាពពេញលេញនៃផ្គរលាន់ទាំងប្រាំពីរនេះ ដែលបានអនុវត្តមកលើអ្នករាល់គ្នាដោយគ្មានក្តីមេត្តា អ្នកនឹងយល់ពីឫទ្ធានុភាពពេញលេញនៃសេចក្តីក្រោធរបស់ខ្ញុំ ដែលត្រូវបញ្ចេញតាមបញ្ជារបស់ខ្ញុំ នៅក្នុងអំណាចពេញលេញនៃផ្គរលាន់ទាំងប្រាំពីរ ដែលនឹងបំផ្លាញ អាមេរូកា (AmeruKa) សូម្បីតែប្រជាជាតិស្រីពេស្យាដ៏គួរឲ្យស្អប់ខ្ពើមរបស់អ្នក និងផ្លូវសហាយស្មន់ទាំងអស់របស់នាង!!!

សំឡេងនៃផ្គរលាន់ទាំងប្រាំពីរមានបន្ទូល!!!

ហើយដូច្នេះហើយ ទើបខ្ញុំ យេស៊ូវ ហាម៉ាស៊ីអា ដែលឥឡូវនេះបញ្ចេញព្រះបន្ទូលនៃអំណាចដែលបានធ្វើឱ្យសកម្មទាំងនេះ ទៅក្នុងផែនដី ដែលមានរសជាតិផ្អែមដូចទឹកឃ្មុំនៅក្នុងមាត់ សម្រាប់អ្នកដែលមានជំនឿស្មោះត្រង់ក្នុងព្រះយេស៊ូវគ្រីស្ទ ហើយត្រូវបានប្រទានយ៉ាងអស្ចារ្យពីព្រះហឫទ័យរបស់ អដូណាយ-យ៉ូវ៉េ ដោយសេចក្តីស្រឡាញ់ដ៏អស្ចារ្យបំផុត ប៉ុន្តែត្រូវបានថ្លែងនៅក្នុងភាពល្វីងជូរចត់នៃទឹកប្រមាត់ សម្រាប់ការជំនុំជម្រះដែលត្រូវបញ្ចេញយ៉ាងពេញលេញ ដែលឃោរឃៅ ធ្ងន់ធ្ងរ ហើយនឹងត្រូវបានប្រទានដោយគ្មានសេចក្តីមេត្តាករុណាទាល់តែសោះ ទោះបីជាអណ្តាតដ៏ស្មោកគ្រោករបស់អ្នកនឹងអង្វរក៏ដោយ ក៏អ្នកនឹងមិនអាចគេចផុតឡើយ!!!

ហេតុដូច្នេះហើយ ចូរយល់ពីអ្វីដែលនឹងមកដល់លើផែនដីឥឡូវនេះ ដែលនឹងបំផ្លាញប្រជាជាតិទាំងឡាយយ៉ាងដាច់ខាត នៅក្នុងសេចក្តីក្រោធពេញលេញ។ ដូច្នេះហើយ ខ្ញុំ យេស៊ូវ ហាម៉ាស៊ីអា សូមបញ្ចេញព្រះបន្ទូលនៃបញ្ជាទាំងនេះ ពីជម្រៅខាងក្នុងនៃ ប៊ីប្លារីឌាន១៤៤ ដោយព្រះវិញ្ញាណនៃការព្យាករណ៍យ៉ាងពេញលេញ តាមការបញ្ជារបស់ អដូណាយ-យ៉ូវ៉េ ដែលស្បថ ប្រកាស និងធ្វើឱ្យសកម្មយ៉ាងពេញលេញចូលទៅក្នុងផែនដី នូវវិវរណៈ ជំពូក ១១ ៖

នោះមានគេឲ្យបំពង់រុក្ខជាតិ១ មកខ្ញុំ ដូចជាដំបង ហើយទេវតាឈរប្រាប់ថា ចូរក្រោកឡើង វាស់ព្រះវិហារនៃព្រះ និងអាសនៈ ព្រមទាំងពួកអ្នកដែលថ្វាយបង្គំនៅទីនោះផង។ ឯទីលាននៅខាងក្រៅព្រះវិហារ នោះចូរទុកចោលចេញ កុំវាស់ឡើយ ព្រោះទីនោះបានប្រគល់ឲ្យសាសន៍ដទៃហើយ គេនឹងជាន់ឈ្លីទីក្រុងបរិសុទ្ធ អស់៤២ខែ។

ហេតុដូច្នេះ សូមដឹងចុះអ្នករាល់គ្នា ដែលអាន និងចំអកដល់ព្រះបន្ទូលទាំងនេះ ដែលប្រទានដោយព្រះហស្តនៃអ្នកនាំសារ និងអ្នកបម្រើដ៏មានឫទ្ធិរបស់ខ្ញុំ៖ អ្នករាល់គ្នានឹងយល់ថា អ្នកគ្រាន់តែជាធូលីដីនៅក្នុងខ្សែព្រះនេត្ររបស់ អដូណាយ-យ៉ូវ៉េ ដែលបណ្តាលឱ្យអ្វីៗនៅលើផែនដី លែងមានទៀត នៅពេលដែលត្រូវបានយកទៅក្នុងការជំនុំជម្រះដ៏ក្រោធខឹង អស់កល្បជានិច្ច។ ហេតុដូច្នេះ ចូរយល់ថា ប្រសិនបើមិនមែនដោយសារ ព្រះលោហិតរបស់កូនចៀមនៃ ព្រះយេហូវ៉ា ទេ អ្នករាល់គ្នានឹងដូចជា សូដុំម និង ហ្គូម៉ូរ៉ា ដោយគ្មានក្តីមេត្តាករុណា នៅក្នុងសេចក្តីក្រោធដ៏ពេញលេញ និងហិង្សា មកលើពួកដែលបានធ្លាក់ចុះ។ ប៉ុន្តែ វាត្រូវបានសរសេរទុកមកថា៖ អដូណាយ-យ៉ូវ៉េ ស្រឡាញ់ពិភពលោកដល់ម៉្លេះ បានជាទ្រង់ប្រទានព្រះរាជបុត្រាតែមួយរបស់ទ្រង់ ដើម្បីឱ្យអ្នកណាដែលជឿលើខ្ញុំ មិនវិនាសឡើយ តែមានជីវិតអស់កល្បជានិច្ច។

ហើយដូច្នេះ សេចក្តីក្រោធរបស់កូនចៀម នឹងមកលើផែនដីទាំងមូល នៅក្នុងការជំនុំជម្រះតាមគោលដៅ ដ៏ពេញលេញ ហិង្សា ពិតប្រាកដ និងជាក់លាក់ ដែលនឹងចេតនាបណ្ដេញអ្នកដែលខ្ចាត់ខ្ចាយ និងខ្ចាត់ព្រាត់ពីដែនដីសន្យា ទៅកាន់អ៊ីស្រាអែល ដែល ហេណុក ជាអ្នកកត់ត្រា និង អេលីយ៉ា ជាសាសន៍ធីសប៊ីត នឹងថែរក្សាពួកគេនៅក្រោមស្លាបដ៏មានអំណាចរបស់ពួកគេ និងតាមរយៈការវាយផ្ចាលយ៉ាងខ្លាំង។ ទោះជាយ៉ាងណាក៏ដោយ ត្រូវបានផ្តល់ដោយក្តីថ្នាក់ថ្នម ដល់កូនចៅដែលខ្វាក់ភ្នែកមួយផ្នែករបស់ខ្ញុំ ដែលនឹង មើលឃើញ ម្តងទៀត ហើយត្រូវបានប្រោសលោះដោយអ្នកប្រោសលោះសាច់ញាតិរបស់ពួកគេ។ ហើយត្រូវបាននាំទៅអ៊ីស្រាអែល ដើម្បីអង្គុយនៅជិតជើងរបស់ពួកគេ នៅក្នុងភាពស្ងប់ស្ងាត់ និងគួរឱ្យគោរពកោតខ្លាច ដ្បិតពួកគេជាចាស់ទុំលើអ្នករាល់គ្នា លើកលែងតែពួក ១៤៤,០០០ របស់ខ្ញុំ ដែលនឹងយាមកាម និងការពារអ្នករាល់គ្នា ហើយនឹងឈរយាមយ៉ាងច្បាស់លាស់ ពេញលេញ និងខ្លាំងក្លា លើអ៊ីស្រាអែល ដែលជាតំបន់សុវត្ថិភាពរួចទៅហើយ ហើយទីក្រុងជម្រករបស់ខ្ញុំឥឡូវនេះ ត្រូវបានបង្កើតឡើងដោយសាក្សីទាំងពីររបស់ខ្ញុំ ហើយពួកគេបានត្រៀមខ្លួនជាស្រេចសម្រាប់ការស្រែក និងសំឡេងត្រែដ៏ខ្លាំងក្លានោះ។ សេឡា!
